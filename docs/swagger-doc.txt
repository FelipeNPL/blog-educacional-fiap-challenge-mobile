Schemas
CreatePostDto{
title*	string
example: O Futuro da Inteligência Artificial na Educação
O título da postagem

content*	string
example: Conteúdo detalhado sobre IA...
O conteúdo da postagem

author	string
example: João Silva
O autor da postagem (opcional)

}
UpdatePostDto{
title	string
example: O Futuro da IA na Educação (Atualizado)
O novo título da postagem

content	string
example: Novo conteúdo atualizado...
O novo conteúdo da postagem

}
RegisterDto{
email*	string
example: usuario@exemplo.com
E-mail do novo usuário

password*	string
example: senha123
minLength: 6
Senha do novo usuário (mínimo 6 caracteres)

name*	string
example: Nome do Usuário
Nome completo do usuário

}
LoginDto{
email*	string
example: usuario@exemplo.com
E-mail do usuário

password*	string
example: senha123
Senha do usuário

}
